"""
网络流量采集器 - 使用 Scapy 抓包
对应架构文档 §3.3 流量采集架构
"""
from scapy.all import sniff, IP, TCP, UDP, ICMP
from scapy.packet import Packet
import threading
import queue
from datetime import datetime
from typing import Optional
import logging
 
logger = logging.getLogger(__name__)
 
 
class PacketCollector:
    """实时网络流量采集器"""
 
    def __init__(self, interface: str = None, queue_size: int = 10000):
        """
        Args:
            interface: 网卡名称，None 表示自动选择
            queue_size: 数据包缓冲队列大小
        """
        self.interface = interface
        self.packet_queue = queue.Queue(maxsize=queue_size)
        self._running = False
        self._thread: Optional[threading.Thread] = None
        self._drop_count = 0  # 记录丢弃的包数
 
    def _packet_callback(self, packet: Packet):
        """Scapy 回调函数，解析每个数据包"""
        if not packet.haslayer(IP):
            return
 
        ip_layer = packet[IP]
        parsed = {
            'timestamp': datetime.now().isoformat(),
            'src_ip': ip_layer.src,
            'dst_ip': ip_layer.dst,
            'protocol': 'OTHER',
            'src_port': 0,
            'dst_port': 0,
            'packet_len': len(packet),
            'ttl': ip_layer.ttl,
            'tcp_flags': '',
        }
 
        if packet.haslayer(TCP):
            tcp_layer = packet[TCP]
            parsed['protocol'] = 'TCP'
            parsed['src_port'] = tcp_layer.sport
            parsed['dst_port'] = tcp_layer.dport
            parsed['tcp_flags'] = str(tcp_layer.flags)
        elif packet.haslayer(UDP):
            udp_layer = packet[UDP]
            parsed['protocol'] = 'UDP'
            parsed['src_port'] = udp_layer.sport
            parsed['dst_port'] = udp_layer.dport
        elif packet.haslayer(ICMP):
            parsed['protocol'] = 'ICMP'
 
        try:
            self.packet_queue.put_nowait(parsed)
        except queue.Full:
            self._drop_count += 1
            if self._drop_count % 1000 == 0:
                logger.warning(
                    f"[PacketCollector] 队列已满，已丢弃 {self._drop_count} 个数据包"
                )
 
    def start(self):
        """启动抓包（在后台线程中运行）"""
        self._running = True
        self._thread = threading.Thread(target=self._run, daemon=True)
        self._thread.start()
        logger.info(f"[PacketCollector] 已启动，接口: {self.interface or '自动'}")
 
    def _run(self):
        sniff(prn=self._packet_callback, iface=self.interface,
              store=False, stop_filter=lambda _: not self._running)
 
    def stop(self):
        """停止抓包"""
        self._running = False
        if self._thread:
            self._thread.join(timeout=5)
        logger.info("[PacketCollector] 已停止")
 
    def get_packets(self, max_count: int = 100) -> list:
        """从队列中取出数据包"""
        packets = []
        for _ in range(max_count):
            try:
                packets.append(self.packet_queue.get_nowait())
            except queue.Empty:
                break
        return packets

"""
日志采集器 - 读取系统日志和 Web 日志
对应架构文档 §3.2 数据源
"""
import os
import re
from datetime import datetime
from typing import Iterator, Optional
import logging
 
logger = logging.getLogger(__name__)
 
 
class LogCollector:
    """日志文件采集器（支持增量读取）"""
 
    # Apache/Nginx access.log 正则
    ACCESS_LOG_PATTERN = re.compile(
        r'(?P<ip>\S+) \S+ \S+ \[(?P<time>[^\]]+)\] '
        r'"(?P<method>\S+) (?P<url>\S+) \S+" '
        r'(?P<status>\d+) (?P<size>\d+)'
    )
 
    def __init__(self, log_path: str, log_type: str = 'web'):
        """
        Args:
            log_path: 日志文件路径
            log_type: 日志类型 'web' | 'system' | 'app'
        """
        self.log_path = log_path
        self.log_type = log_type
        self._last_position = 0  # 记录上次读取位置（增量读取）
 
    def read_new_lines(self) -> list:
        """读取自上次以来的新增行"""
        if not os.path.exists(self.log_path):
            return []
 
        results = []
        try:
            with open(self.log_path, 'r', encoding='utf-8', errors='ignore') as f:
                f.seek(self._last_position)
                for line in f:
                    parsed = self._parse_line(line.strip())
                    if parsed:
                        results.append(parsed)
                self._last_position = f.tell()
        except IOError as e:
            logger.error(f"[LogCollector] 读取日志失败: {e}")
        return results
 
    def _parse_line(self, line: str) -> Optional[dict]:
        """解析单行日志"""
        if self.log_type == 'web':
            return self._parse_web_log(line)
        elif self.log_type == 'system':
            return self._parse_system_log(line)
        return None
 
    def _parse_web_log(self, line: str) -> Optional[dict]:
        """解析 Web access.log"""
        match = self.ACCESS_LOG_PATTERN.match(line)
        if not match:
            return None
        return {
            'timestamp': match.group('time'),
            'src_ip': match.group('ip'),
            'http_method': match.group('method'),
            'url': match.group('url'),
            'status_code': int(match.group('status')),
            'response_size': int(match.group('size')),
            'log_type': 'web',
        }
 
    def _parse_system_log(self, line: str) -> Optional[dict]:
        """解析系统日志（/var/log/auth.log 等）"""
        parts = line.split(maxsplit=3)
        if len(parts) < 4:
            return None
        return {
            'timestamp': f"{parts[0]} {parts[1]} {parts[2]}",
            'process': parts[3].split(':')[0] if ':' in parts[3] else parts[3],
            'message': parts[3] if len(parts) > 3 else '',
            'log_type': 'system',
        }

"""
威胁情报采集器 - 从外部 API 获取 IOC
对应架构文档 §3.2 威胁情报数据源
【安全修复】增加请求超时、异常处理、缓存TTL管理、速率限制
"""
import requests
from typing import Set, Optional
from datetime import datetime, timedelta
import logging
import threading
 
logger = logging.getLogger(__name__)
 
 
class ThreatIntelCollector:
    """威胁情报采集器"""
 
    # IP 地址验证正则
    IP_PATTERN = None  # 延迟导入 re
 
    def __init__(self, abuseipdb_key: str = '', virustotal_key: str = ''):
        self.abuseipdb_key = abuseipdb_key
        self.virustotal_key = virustotal_key
        self._ip_blacklist: Set[str] = set()
        self._domain_blacklist: Set[str] = set()
        self._last_update: Optional[datetime] = None
        self._cache_ttl = timedelta(hours=1)  # 缓存1小时
        self._lock = threading.Lock()  # 线程安全锁
        self._request_interval = 1.0  # API 请求最小间隔（秒）
        self._last_request_time: Optional[datetime] = None
 
    @staticmethod
    def _validate_ip(ip: str) -> bool:
        """验证 IP 地址格式，防止注入"""
        import re
        parts = ip.split('.')
        if len(parts) != 4:
            return False
        try:
            return all(0 <= int(p) <= 255 for p in parts)
        except ValueError:
            return False
 
    def check_ip(self, ip: str) -> dict:
        """
        检查 IP 是否在威胁情报黑名单中
        对应架构文档 §5.2 前置规则过滤
        """
        # 【安全修复】先验证 IP 格式
        if not self._validate_ip(ip):
            logger.warning(f"[ThreatIntel] 无效的 IP 地址格式: {ip}")
            return {'is_malicious': False, 'source': 'none', 'reason': 'invalid_ip_format'}
 
        # 先查本地缓存
        with self._lock:
            if ip in self._ip_blacklist:
                return {'is_malicious': True, 'source': 'cache'}
 
        # 查 AbuseIPDB
        if self.abuseipdb_key:
            result = self._query_abuseipdb(ip)
            if result:
                return result
 
        return {'is_malicious': False, 'source': 'none'}
 
    def _query_abuseipdb(self, ip: str) -> Optional[dict]:
        """查询 AbuseIPDB API"""
        try:
            resp = requests.get(
                'https://api.abuseipdb.com/api/v2/check',
                headers={'Key': self.abuseipdb_key, 'Accept': 'application/json'},
                params={'ipAddress': ip, 'maxAgeInDays': 30},
                timeout=5  # 【安全修复】增加超时
            )
            resp.raise_for_status()  # 【安全修复】检查HTTP错误
            data = resp.json()
            if data.get('data', {}).get('abuseConfidenceScore', 0) > 50:
                with self._lock:
                    self._ip_blacklist.add(ip)
                return {'is_malicious': True, 'source': 'abuseipdb',
                        'score': data['data']['abuseConfidenceScore']}
        except requests.exceptions.Timeout:
            logger.error("[ThreatIntel] AbuseIPDB 查询超时")
        except requests.exceptions.RequestException as e:
            logger.error(f"[ThreatIntel] AbuseIPDB 查询失败: {e}")
        return None
 
    def check_domain(self, domain: str) -> dict:
        """检查域名是否在黑名单中"""
        if not domain or not isinstance(domain, str):
            return {'is_malicious': False, 'source': 'none'}
 
        with self._lock:
            if domain.lower() in self._domain_blacklist:
                return {'is_malicious': True, 'source': 'cache'}
 
        return {'is_malicious': False, 'source': 'none'}
 
    def update_blacklist(self):
        """定时更新黑名单"""
        with self._lock:
            self._last_update = datetime.now()
            count = len(self._ip_blacklist)
        logger.info(f"[ThreatIntel] 黑名单已更新，当前 {count} 条")
 
    def get_blacklist_stats(self) -> dict:
        """获取黑名单统计信息"""
        with self._lock:
            return {
                'ip_count': len(self._ip_blacklist),
                'domain_count': len(self._domain_blacklist),
                'last_update': self._last_update.isoformat() if self._last_update else None,
            }

"""
网络流量特征提取器
对应架构文档 §4.2.1 网络流量特征
"""
import numpy as np
from collections import defaultdict
from typing import List, Dict
import logging
 
logger = logging.getLogger(__name__)
 
 
class FlowFeatureExtractor:
    """基于五元组的流统计特征提取"""
 
    def __init__(self, timeout: float = 120.0):
        """
        Args:
            timeout: 流超时时间（秒），超过此时间视为新流
        """
        self.timeout = timeout
        self._flows: Dict[str, dict] = {}  # 五元组 -> 流状态
 
    def _make_flow_key(self, packet: dict) -> str:
        """生成五元组 key"""
        return f"{packet['src_ip']}:{packet['src_port']}-{packet['dst_ip']}:{packet['dst_port']}-{packet['protocol']}"
 
    def extract(self, packets: List[dict]) -> List[dict]:
        """
        从数据包列表中提取流特征
        返回特征字典列表
        """
        features_list = []
        for pkt in packets:
            key = self._make_flow_key(pkt)
            if key not in self._flows:
                self._flows[key] = {
                    'start_time': pkt['timestamp'],
                    'packets': [],
                    'src_ip': pkt['src_ip'],
                    'dst_ip': pkt['dst_ip'],
                    'protocol': pkt['protocol'],
                }
            self._flows[key]['packets'].append(pkt)
 
        # 计算每个流的统计特征
        for key, flow in self._flows.items():
            pkts = flow['packets']
            if len(pkts) < 2:
                continue
 
            pkt_lens = [p['packet_len'] for p in pkts]
            features = {
                'src_ip': flow['src_ip'],
                'dst_ip': flow['dst_ip'],
                'protocol': flow['protocol'],
                'flow_pkt_count': len(pkts),
                'flow_byte_count': sum(pkt_lens),
                'pkt_len_mean': np.mean(pkt_lens),
                'pkt_len_std': np.std(pkt_lens),
                'pkt_len_max': max(pkt_lens),
                'pkt_len_min': min(pkt_lens),
                'src_port': pkts[0]['src_port'],
                'dst_port': pkts[0]['dst_port'],
            }
 
            # TCP 标志位计数
            syn_count = sum(1 for p in pkts if 'S' in p.get('tcp_flags', ''))
            ack_count = sum(1 for p in pkts if 'A' in p.get('tcp_flags', ''))
            fin_count = sum(1 for p in pkts if 'F' in p.get('tcp_flags', ''))
            rst_count = sum(1 for p in pkts if 'R' in p.get('tcp_flags', ''))
            features.update({
                'syn_count': syn_count,
                'ack_count': ack_count,
                'fin_count': fin_count,
                'rst_count': rst_count,
            })
 
            features_list.append(features)
 
        # 清理已处理的流
        self._flows.clear()
        return features_list

"""

对应架构文档 §4.2.2 Web 请求特征

from urllib.parse import unquote
import logging
 
logger = logging.getLogger(__name__)
 
 
class WebFeatureExtractor:
    """HTTP 请求特征提取"""
 
    # SQL 注入关键字
    SQL_KEYWORDS = re.compile(
        r'(union|select|insert|update|delete|drop|alter|exec|execute|'
        r'xp_cmdshell|0x|char\(|concat|information_schema)', re.IGNORECASE
    )
    # XSS 关键字
    XSS_KEYWORDS = re.compile(
        r'(<script|javascript:|onerror|onload|alert\(|document\.|'
        r'window\.|eval\()', re.IGNORECASE
    )
    # 命令注入关键字
    CMD_KEYWORDS = re.compile(
        r'(;|\||&&|\$\(|`|bash|sh|nc |ncat|wget|curl|ping|'
        r'passwd|shadow|/etc/)', re.IGNORECASE
    )
    # 特殊字符
    SPECIAL_CHARS = re.compile(r"[?'"<>;|&`$\\{}\[\]()]")
 
    @staticmethod
    def _decode_url(url: str) -> str:
        """
        【安全修复】多层 URL 解码，防止编码绕过
        攻击者可能使用多层编码绕过检测：%253C -> %3C -> <
        """
        decoded = url
        for _ in range(5):  # 最多解码5层，防止无限循环
            try:
                new_decoded = unquote(decoded)
                if new_decoded == decoded:
                    break
                decoded = new_decoded
            except Exception:
                break
        return decoded
 
    def extract(self, log_entry: dict) -> dict:
        """从 Web 日志条目中提取特征"""
        url = log_entry.get('url', '')
        method = log_entry.get('http_method', 'GET')
 
        # 【安全修复】对 URL 进行多层解码后再提取特征
        decoded_url = self._decode_url(url)
 
        features = {
            'url_length': len(url),
            'decoded_url_length': len(decoded_url),
            'http_method': method,
            'param_count': url.count('&') + url.count(';') + 1,
            'status_code': log_entry.get('status_code', 200),
            'response_size': log_entry.get('response_size', 0),
            'special_char_count': len(self.SPECIAL_CHARS.findall(decoded_url)),
            'sql_keyword_count': len(self.SQL_KEYWORDS.findall(decoded_url)),
            'xss_keyword_count': len(self.XSS_KEYWORDS.findall(decoded_url)),
            'cmd_keyword_count': len(self.CMD_KEYWORDS.findall(decoded_url)),
            # 保留原始 URL 和解码后的 URL 供检测引擎使用
            'url_raw': url,
            'url_decoded': decoded_url,
        }
 
        return features

"""
特征处理流水线
对应架构文档 §4.3 特征处理流水线
"""
import joblib
import numpy as np
from sklearn.preprocessing import StandardScaler, MinMaxScaler, LabelEncoder
from typing import List, Dict, Optional
import logging
 
logger = logging.getLogger(__name__)
 
 
class FeaturePipeline:
    """特征标准化与编码流水线"""
 
    def __init__(self):
        self.scaler = StandardScaler()
        self.label_encoders = {}  # 列名 -> LabelEncoder
        self.numeric_columns = []
        self.categorical_columns = []
        self._fitted = False
 
    def fit(self, features_list: List[dict]):
        """从训练数据中拟合标准化参数"""
        import pandas as pd
        df = pd.DataFrame(features_list)
 
        # 自动区分数值列和类别列
        self.numeric_columns = df.select_dtypes(include=[np.number]).columns.tolist()
        self.categorical_columns = df.select_dtypes(include=['object']).columns.tolist()
 
        # 拟合数值标准化
        if self.numeric_columns:
            self.scaler.fit(df[self.numeric_columns].values)
 
        # 拟合类别编码
        for col in self.categorical_columns:
            le = LabelEncoder()
            le.fit(df[col].astype(str))
            self.label_encoders[col] = le
 
        self._fitted = True
 
    def transform(self, features_list: List[dict]) -> np.ndarray:
        """将特征字典列表转换为标准化后的 numpy 数组"""
        import pandas as pd
        df = pd.DataFrame(features_list)
 
        # 数值标准化
        if self.numeric_columns:
            numeric_vals = self.scaler.transform(df[self.numeric_columns].values)
        else:
            numeric_vals = np.empty((len(df), 0))
 
        # 类别编码
        cat_vals = []
        for col in self.categorical_columns:
            if col in self.label_encoders:
                # 处理未见过的类别
                known_classes = set(self.label_encoders[col].classes_)
                encoded = df[col].astype(str).apply(
                    lambda x: self.label_encoders[col].transform([x])[0]
                    if x in known_classes else -1
                ).values
                cat_vals.append(encoded.reshape(-1, 1))
 
        if cat_vals:
            cat_array = np.hstack(cat_vals)
            return np.hstack([numeric_vals, cat_array])
        return numeric_vals
 
    def save(self, path: str):
        """保存流水线参数"""
        joblib.dump(self, path)
 
    @staticmethod
    def load(path: str) -> 'FeaturePipeline':
        """加载流水线参数"""
        return joblib.load(path)

import pandas as pd

# NSL-KDD 列名
COLUMNS = [
    'duration', 'protocol_type', 'service', 'flag', 'src_bytes', 'dst_bytes',
    'land', 'wrong_fragment', 'urgent', 'hot', 'num_failed_logins', 'logged_in',
    'num_compromised', 'root_shell', 'su_attempted', 'num_root', 'num_file_creations',
    'num_shells', 'num_access_files', 'num_outbound_cmds', 'is_host_login',
    'is_guest_login', 'count', 'srv_count', 'serror_rate', 'srv_serror_rate',
    'rerror_rate', 'srv_rerror_rate', 'same_srv_rate', 'diff_srv_rate',

    'dst_host_same_srv_rate', 'dst_host_diff_srv_rate',
    'dst_host_same_src_port_rate', 'dst_host_srv_diff_host_rate',
    'dst_host_serror_rate', 'dst_host_srv_serror_rate',
    'dst_host_rerror_rate', 'dst_host_srv_rerror_rate', 'label', 'difficulty'
]
 
 
def load_data(path: str) -> pd.DataFrame:
    """加载 NSL-KDD 数据集"""
    df = pd.read_csv(path, header=None, names=COLUMNS)
    return df
 
 
def preprocess(df: pd.DataFrame):
    """数据预处理"""
    # 将标签简化为 Normal / Attack
    df['target'] = df['label'].apply(lambda x: 'normal' if x == 'normal' else 'attack')
 
    # 类别编码
    cat_cols = ['protocol_type', 'service', 'flag']
    for col in cat_cols:
        df[col] = pd.factorize(df[col])[0]
 
    # 提取特征和标签
    feature_cols = [c for c in COLUMNS if c not in ['label', 'difficulty', 'target']]
    X = df[feature_cols].values
    y = df['target'].values
    return X, y
 
 
def train():
    """训练 DDoS 检测模型"""
    print("=" * 50)
    print("DDoS 检测模型训练（随机森林）")
    print("=" * 50)
 
    # 加载数据
    train_df = load_data('data/datasets/KDDTrain+.txt')
    test_df = load_data('data/datasets/KDDTest+.txt')
 
    X_train, y_train = preprocess(train_df)
    X_test, y_test = preprocess(test_df)
 
    print(f"训练集: {X_train.shape}, 测试集: {X_test.shape}")
 
    # 训练随机森林
    model = RandomForestClassifier(
        n_estimators=100,
        max_depth=20,
        n_jobs=-1,
        random_state=42,
    )
    model.fit(X_train, y_train)
 
    # 评估
    y_pred = model.predict(X_test)
    acc = accuracy_score(y_test, y_pred)
    print(f"\n准确率: {acc:.4f}")
    print("\n分类报告:")
    print(classification_report(y_test, y_pred))
 
    # 保存模型
    model_dir = 'models/saved'
    os.makedirs(model_dir, exist_ok=True)
    model_path = os.path.join(model_dir, 'ddos_rf_v1.pkl')
    joblib.dump(model, model_path)
    print(f"\n模型已保存至: {model_path}")
 
 
if __name__ == '__main__':
    train()

        ['xss'] * len(XSS_SAMPLES) +
        ['command_injection'] * len(CMD_INJECTION_SAMPLES) +
        ['path_traversal'] * len(PATH_TRAVERSAL_SAMPLES) +

    print(f"  正常: {len(NORMAL_SAMPLES)}")

    )

    model = MultinomialNB(alpha=0.1)
 
    # 组成 Pipeline
    pipeline = Pipeline([
        ('tfidf', vectorizer),
        ('clf', model),
    ])
 
    # 训练
    pipeline.fit(X_train, y_train)
 
    # 评估
    y_pred = pipeline.predict(X_test)
    print("\n分类报告:")
    print(classification_report(y_test, y_pred))
 
    # 交叉验证
    cv_scores = cross_val_score(pipeline, texts, labels, cv=5, scoring='f1_weighted')
    print(f"\n5折交叉验证 F1 分数: {cv_scores.mean():.4f} (+/- {cv_scores.std():.4f})")
 
    # 对抗性测试（编码绕过）
    adversarial_tests = [
        ("%27 OR 1%3D1--", "sql_injection"),
        ("%3Cscript%3Ealert(1)%3C/script%3E", "xss"),

        ("/api/users?page=1", "normal"),

    model_path = os.path.join(model_dir, 'web_attack_nb_v1.pkl')
    joblib.dump(pipeline, model_path)
    print(f"\n模型已保存至: {model_path}")

"""

import numpy as np

from sklearn.preprocessing import StandardScaler
from sklearn.metrics import classification_report

import os
 
 
def generate_normal_data(n_samples: int = 5000) -> np.ndarray:
    """模拟正常网络行为数据"""
    np.random.seed(42)
    data = np.column_stack([
        np.random.normal(50, 10, n_samples),    # 平均包长
        np.random.normal(100, 30, n_samples),    # 流字节数
        np.random.normal(10, 3, n_samples),      # 流包数
        np.random.normal(5, 2, n_samples),       # 不同目的端口数
        np.random.normal(0.1, 0.05, n_samples),  # SYN 比率
    ])
    return data
 
 
def generate_anomaly_data(n_samples: int = 200) -> np.ndarray:
    """模拟异常网络行为数据"""
    np.random.seed(123)
    anomalies = np.column_stack([
        np.random.uniform(200, 1500, n_samples),   # 异常大包
        np.random.uniform(5000, 50000, n_samples),  # 异常大流量
        np.random.uniform(500, 5000, n_samples),    # 异常多包
        np.random.uniform(50, 500, n_samples),      # 异常多端口
        np.random.uniform(0.5, 1.0, n_samples),     # 异常高 SYN 比率
    ])
    return anomalies
 
 
def train_isolation_forest():
    """训练 Isolation Forest 模型"""
    print("=" * 50)
    print("异常行为检测模型训练（Isolation Forest）")
    print("=" * 50)
 
    # 用正常数据训练
    X_normal = generate_normal_data()
    print(f"训练数据（正常行为）: {X_normal.shape}")
 
    model = IsolationForest(
        n_estimators=100,
        contamination=0.01,   # 假设异常占 1%
        random_state=42,
        n_jobs=-1,
    )
    model.fit(X_normal)
 
    # 测试
    X_test_normal = generate_normal_data(n_samples=500)
    X_test_anomaly = generate_anomaly_data(n_samples=200)
 
    pred_normal = model.predict(X_test_normal)
    pred_anomaly = model.predict(X_test_anomaly)
 
    normal_accuracy = (pred_normal == 1).mean()
    anomaly_detection_rate = (pred_anomaly == -1).mean()
 
    print(f"\n正常样本正确识别率: {normal_accuracy:.4f}")
    print(f"异常样本检测率: {anomaly_detection_rate:.4f}")
 
    # 保存
    model_dir = 'models/saved'
    os.makedirs(model_dir, exist_ok=True)
    model_path = os.path.join(model_dir, 'anomaly_if_v1.pkl')
    joblib.dump(model, model_path)
    print(f"\n模型已保存至: {model_path}")
    return model
 
 
def train_autoencoder():
    """
    【新增】训练 Autoencoder 异常检测模型
    对应架构文档 §5.4.4 模型二
    """
    print("\n" + "=" * 50)
    print("异常行为检测模型训练（Autoencoder）")
    print("=" * 50)
 
    import tensorflow as tf
    from tensorflow import keras
    from tensorflow.keras import layers
 
    # 生成训练数据（仅正常数据）
    X_normal = generate_normal_data(n_samples=5000)
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X_normal)
 
    # 构建 Autoencoder
    input_dim = X_scaled.shape[1]
 
    autoencoder = keras.Sequential([
        # 编码器
        layers.Dense(16, activation='relu', input_shape=(input_dim,)),
        layers.Dense(8, activation='relu'),
        layers.Dense(4, activation='relu'),  # 瓶颈层
        # 解码器
        layers.Dense(8, activation='relu'),
        layers.Dense(16, activation='relu'),
        layers.Dense(input_dim, activation='linear'),
    ])
 
    autoencoder.compile(optimizer='adam', loss='mse')
 
    # 训练
    history = autoencoder.fit(
        X_scaled, X_scaled,
        epochs=50,

    mse_anomaly = np.mean(np.power(X_test_anomaly - recon_anomaly, 2), axis=1)

def preprocess(df: pd.DataFrame):
    """数据预处理"""

"""
检测引擎基类
"""
from abc import ABC, abstractmethod
from typing import Optional
from dataclasses import dataclass, field
 
 
@dataclass
class DetectionResult:
    """检测结果"""
    threat_type: str       # 威胁类型: 'ddos', 'intrusion', 'web_attack', 'anomaly', 'normal'
    threat_level: str      # 威胁等级: 'low', 'medium', 'high', 'critical'
    confidence: float      # 置信度 0.0 ~ 1.0
    details: str           # 详细描述
    source_ip: str = ''    # 来源 IP
    raw_data: dict = field(default_factory=dict)  # 原始数据
 
 
class BaseDetector(ABC):
    """检测引擎基类"""
 
    @abstractmethod
    def detect(self, features) -> Optional[DetectionResult]:
        """执行检测，返回结果或 None（无法判定）"""
        pass
 
    @abstractmethod
    def load_model(self, model_path: str):
        """加载模型"""
        pass
 
    @property
    def is_ready(self) -> bool:
        """检测引擎是否已就绪"""
        return True

            self.model = joblib.load(model_path)
            self.scaler = joblib.load(os.path.join(model_dir, 'intrusion_scaler_v1.pkl'))
            self.label_encoder = joblib.load(os.path.join(model_dir, 'intrusion_label_encoder_v1.pkl'))
            self.feature_columns = joblib.load(os.path.join(model_dir, 'intrusion_feature_cols_v1.pkl'))
            logger.info(f"[IntrusionDetector] 随机森林模型已加载: {model_path}")

    def is_ready(self) -> bool:
        return self.model is not None and self.scaler is not None

            return None
 
        # 构造特征向量
        feature_vector = []

            feature_vector.append(features.get(col, 0))

        threat_level = level_map.get(prediction, 'medium')

            threat_type='intrusion',
            threat_level=threat_level,
            confidence=confidence,
            details=f'入侵检测: {prediction}',
            source_ip=features.get('src_ip', '')
        )

"""
Web 攻击检测引擎（规则引擎 + ML 双重策略）
对应架构文档 §5.4.3
【安全修复】增加 URL 解码防绕过，增加路径遍历检测规则
"""
import re
import joblib
from urllib.parse import unquote
from .base import BaseDetector, DetectionResult
import logging
 
logger = logging.getLogger(__name__)
 
 
class WebAttackDetector(BaseDetector):
    """Web 攻击检测引擎"""
 
    # 规则引擎的正则模式（快速通道）
    RULES = {
        'sql_injection': re.compile(
            r"(\b(union\s+select|select\s+.+\s+from|insert\s+into|drop\s+table|"
            r"or\s+1\s*=\s*1|'\s*or\s+'|;\s*delete|;\s*drop|xp_cmdshell|"
            r"information_schema|sleep\s*\(|benchmark\s*\().*\b)",
            re.IGNORECASE
        ),
        'xss': re.compile(
            r"(<\s*script|javascript\s*:|on(error|load|click|mouseover)\s*=|"
            r"alert\s*\(|document\.(cookie|domain|write)|eval\s*\()",
            re.IGNORECASE
        ),
        'command_injection': re.compile(
            r"(;\s*(cat|ls|id|whoami|uname|pwd|wget|curl|nc|bash|sh|python|perl)\b|"
            r"\|\s*(cat|ls|id|whoami)|`\s*(cat|ls|id))",
            re.IGNORECASE
        ),
        'path_traversal': re.compile(
            r"(\.\./|\.\.\\|%2e%2e|%252e|/etc/passwd|/etc/shadow|/proc/self)",
            re.IGNORECASE
        ),
    }
 
    def __init__(self):
        self.ml_model = None
 
    def load_model(self, model_path: str):
        self.ml_model = joblib.load(model_path)
        logger.info(f"[WebDetector] ML 模型已加载: {model_path}")
 
    @staticmethod
    def _decode_url(url: str) -> str:
        """
        【安全修复】多层 URL 解码，防止编码绕过
        """
        decoded = url
        for _ in range(5):
            try:
                new_decoded = unquote(decoded)
                if new_decoded == decoded:
                    break
                decoded = new_decoded
            except Exception:
                break
        return decoded
 
    def detect(self, features: dict) -> DetectionResult:
        url = features.get('url_raw', features.get('url', ''))
 
        # 【安全修复】对 URL 进行多层解码后再进行规则匹配
        decoded_url = self._decode_url(url)
 
        # 第一步：规则引擎快速匹配（对解码后的 URL 进行检测）
        for attack_type, pattern in self.RULES.items():
            if pattern.search(decoded_url):
                return DetectionResult(
                    threat_type='web_attack',
                    threat_level='high',
                    confidence=0.95,
                    details=f'规则引擎命中: {attack_type}',
                    source_ip=features.get('src_ip', '')
                )
 
        # 第二步：ML 文本分类（规则未命中时，同时检测原始URL和解码URL）
        if self.ml_model is not None:
            # 对解码后的 URL 进行 ML 检测
            prediction = self.ml_model.predict([decoded_url])[0]
            if prediction != 'normal':
                proba = self.ml_model.predict_proba([decoded_url])[0]
                max_prob = max(proba)
                return DetectionResult(
                    threat_type='web_attack',
                    threat_level='medium' if max_prob > 0.7 else 'low',
                    confidence=float(max_prob),
                    details=f'ML 检测: {prediction}',
                    source_ip=features.get('src_ip', '')
                )
 
        return DetectionResult(
            threat_type='normal', threat_level='low',
            confidence=0.9, details='正常 Web 请求'
        )

            logger.info(f"[AnomalyDetector] Isolation Forest 已加载: {if_path}")
 
        # 加载 Autoencoder
        ae_dir = os.path.join(model_dir, 'anomaly_ae_v1')
        if os.path.exists(ae_dir):
            import tensorflow as tf
            self.ae_model = tf.keras.models.load_model(ae_dir)
            self.ae_scaler = joblib.load(os.path.join(model_dir, 'anomaly_ae_scaler_v1.pkl'))
            self.ae_threshold = joblib.load(os.path.join(model_dir, 'anomaly_ae_threshold_v1.pkl'))
            logger.info(f"[AnomalyDetector] Autoencoder 已加载: {ae_dir}")
 
    @property
    def is_ready(self) -> bool:
        return self.if_model is not None or self.ae_model is not None
 
    def detect(self, features: dict) -> DetectionResult:
        if not self.is_ready:
            return None
 
        # 提取数值特征
        feature_keys = ['pkt_len_mean', 'flow_byte_count', 'flow_pkt_count',
                       'window_unique_dst_port', 'syn_count']
        feature_values = [features.get(k, 0) for k in feature_keys]
        X = np.array([feature_values], dtype=np.float32)
 
        anomaly_score = 0.0
        is_anomaly = False
 
        # Isolation Forest 检测
        if self.if_model is not None:
            if_pred = self.if_model.predict(X)[0]
            if if_pred == -1:
                is_anomaly = True
                anomaly_score += 0.5
 
        # Autoencoder 检测
        if self.ae_model is not None and self.ae_scaler is not None:
            X_scaled = self.ae_scaler.transform(X)
            reconstruction = self.ae_model.predict(X_scaled, verbose=0)
            mse = np.mean(np.power(X_scaled - reconstruction, 2), axis=1)[0]
            if mse > self.ae_threshold:
                is_anomaly = True
                anomaly_score += 0.5
 
        if is_anomaly:
            return DetectionResult(
                threat_type='anomaly',
                threat_level='high' if anomaly_score > 0.7 else 'medium',
                confidence=min(anomaly_score, 1.0),
                details=f'异常行为检测: 综合异常分数={anomaly_score:.2f}',
                source_ip=features.get('src_ip', '')
            )
 
        return DetectionResult(
            threat_type='normal', threat_level='low',
            confidence=0.9, details='正常行为'
        )

"""
融合决策引擎
对应架构文档 §5.5
"""
from typing import List, Optional
from ..detectors.base import DetectionResult
 
 
class FusionEngine:
    """多模型融合决策引擎"""
 
    # 威胁等级权重
    LEVEL_WEIGHTS = {'critical': 1.0, 'high': 0.8, 'medium': 0.5, 'low': 0.2}
 
    def fuse(self, results: List[Optional[DetectionResult]]) -> DetectionResult:
        """
        融合多个检测引擎的结果
        对应架构文档 §5.5 投票/加权融合
        """
        # 过滤掉 None 和 normal 结果
        valid_results = [r for r in results if r is not None and r.threat_type != 'normal']
 
        if not valid_results:
            return DetectionResult(
                threat_type='normal', threat_level='low',
                confidence=1.0, details='所有引擎判定为正常'
            )
 
        # 加权融合
        total_weight = 0
        weighted_confidence = 0
        threat_types = set()
 
        for r in valid_results:
            weight = self.LEVEL_WEIGHTS.get(r.threat_level, 0.5)
            total_weight += weight
            weighted_confidence += r.confidence * weight
            threat_types.add(r.threat_type)
 
        fused_confidence = weighted_confidence / total_weight if total_weight > 0 else 0
 
        # 确定最终威胁等级
        max_level = max(valid_results, key=lambda r: self.LEVEL_WEIGHTS.get(r.threat_level, 0))
 
        # 多引擎同时报警则提升等级
        if len(valid_results) >= 2:
            level_upgrade = {'low': 'medium', 'medium': 'high', 'high': 'critical'}
            final_level = level_upgrade.get(max_level.threat_level, max_level.threat_level)
        else:
            final_level = max_level.threat_level
 
        return DetectionResult(
            threat_type=' + '.join(threat_types),
            threat_level=final_level,
            confidence=round(fused_confidence, 4),
            details=f'融合 {len(valid_results)} 个引擎结果: {", ".join(r.details for r in valid_results)}',
            source_ip=max_level.source_ip
        )

# IP 地址验证正则
IPV4_PATTERN = re.compile(
    r'^(?:(?:25[0-5]|2[0-4]\d|[01]?\d\d?)\.){3}(?:25[0-5]|2[0-4]\d|[01]?\d\d?)$'
)
 
 
def validate_ip(ip: str) -> bool:
    """
    【安全修复】严格验证 IP 地址格式
    防止通过构造恶意 IP 地址实现命令注入
    """
    if not ip or not isinstance(ip, str):
        return False
    return bool(IPV4_PATTERN.match(ip.strip()))
 
 
class SecurityResponder:
    """安全响应引擎"""
 
    def __init__(self, dry_run: bool = True):
        """
        Args:
            dry_run: True 时只打印不执行（开发模式）
        """
        self.dry_run = dry_run
        self._banned_ips = {}  # ip -> 解封时间
 
    def respond(self, result: DetectionResult):
        """根据威胁等级执行响应"""
        if result.threat_type == 'normal':
            return
 
        level = result.threat_level
        logger.warning(
            f"[响应] {level.upper()} - {result.details} "
            f"(置信度: {result.confidence})"
        )
 
        if level == 'low':
            self._log_only(result)
        elif level == 'medium':
            self._log_only(result)
            self._send_alert(result)
        elif level == 'high':
            self._log_only(result)
            self._send_alert(result)
            self._ban_ip(result.source_ip, duration=timedelta(hours=1))
        elif level == 'critical':
            self._log_only(result)
            self._send_alert(result)
            self._ban_ip(result.source_ip, duration=timedelta(days=1))
            self._isolate_host(result.source_ip)
 
    def _log_only(self, result: DetectionResult):
        """记录日志"""
        logger.info(
            f"[安全事件] type={result.threat_type}, level={result.threat_level}, "
            f"ip={result.source_ip}, confidence={result.confidence}, "
            f"detail={result.details}"
        )
 
    def _send_alert(self, result: DetectionResult):
        """发送告警"""
        # TODO: 接入邮件/短信/Webhook
        logger.warning(
            f"[告警] {result.threat_level.upper()}: {result.details} "
            f"| IP: {result.source_ip}"
        )
 
    def _ban_ip(self, ip: str, duration: timedelta):
        """
        【安全修复】封禁 IP - 修复命令注入漏洞
        原始代码: subprocess.run(f"iptables -A INPUT -s {ip} -j DROP".split())
        漏洞: ip 未经验证直接拼接命令，攻击者可注入如 "1.2.3.4; rm -rf /"
        """
        # 1. 严格验证 IP 格式
        if not validate_ip(ip):
            logger.error(f"[防火墙] 无效的 IP 地址，拒绝封禁: {ip}")
            return
 
        # 2. 检查是否已封禁
        if ip in self._banned_ips:
            return
 
        # 3. 使用列表形式传参（不通过 shell 解释），彻底避免命令注入
        cmd = [
            'iptables',
            '-A', 'INPUT',
            '-s', ip,       # 已验证的 IP，安全
            '-j', 'DROP',
        ]
 
        if self.dry_run:
            logger.info(f"[DRY RUN] 执行: {' '.join(cmd)}")
        else:
            try:
                # 【安全修复】使用列表形式传参，不经过 shell
                subprocess.run(cmd, check=True, timeout=5)
                logger.info(f"[防火墙] 已封禁 IP: {ip}，持续 {duration}")
            except subprocess.CalledProcessError as e:
                logger.error(f"[防火墙] 封禁失败: {e}")
            except subprocess.TimeoutExpired:
                logger.error(f"[防火墙] 封禁命令超时: {ip}")
            except FileNotFoundError:
                logger.error("[防火墙] iptables 命令不存在，请确认系统环境")
 
        self._banned_ips[ip] = datetime.now() + duration
 
    def unban_ip(self, ip: str):
        """解封 IP"""
        if not validate_ip(ip):
            logger.error(f"[防火墙] 无效的 IP 地址，拒绝解封: {ip}")
            return

"""
安全审计日志模块
对应架构文档 §7.2 安全日志 + §11.6 日志防篡改
【安全修复】增加日志完整性校验（哈希链），防止日志被篡改
"""
import json
import logging
import os
import hashlib
from datetime import datetime
from typing import Optional
 
 
class SecurityLogger:
    """结构化安全日志记录器（带完整性保护）"""
 
    def __init__(self, log_dir: str = 'logs', enable_integrity: bool = True):
        self.log_dir = log_dir
        self.enable_integrity = enable_integrity
        os.makedirs(log_dir, exist_ok=True)
        self._setup_logger()
        self._last_hash = ''  # 哈希链：上一条日志的哈希
 
    def _setup_logger(self):
        self.logger = logging.getLogger('security')
        self.logger.setLevel(logging.INFO)
 
        # 文件 Handler（JSON 格式，追加模式）
        log_file = os.path.join(self.log_dir, 'security.log')
        handler = logging.FileHandler(log_file, encoding='utf-8')
        handler.setFormatter(logging.Formatter('%(message)s'))
        self.logger.addHandler(handler)
 
        # 控制台 Handler
        console = logging.StreamHandler()
        console.setFormatter(logging.Formatter(
            '[%(asctime)s] %(levelname)s - %(message)s'
        ))
        self.logger.addHandler(console)
 
    def _compute_hash(self, data: str) -> str:
        """计算日志条目的哈希值（用于完整性校验）"""
        if self._last_hash:
            combined = f"{self._last_hash}:{data}"
        else:
            combined = data
        return hashlib.sha256(combined.encode('utf-8')).hexdigest()[:16]
 
    def log_event(self, event_type: str, level: str, details: dict,
                  source_ip: str = '', confidence: float = 0.0):
        """记录安全事件"""
        event = {
            'timestamp': datetime.now().isoformat(),
            'event_type': event_type,
            'level': level,
            'source_ip': source_ip,
            'confidence': confidence,
            'details': details,
        }
 
        event_str = json.dumps(event, ensure_ascii=False)
 
        # 【安全修复】添加哈希链校验
        if self.enable_integrity:
            event_hash = self._compute_hash(event_str)
            event['integrity'] = {
                'hash': event_hash,
                'prev_hash': self._last_hash or 'genesis',
                'algorithm': 'sha256',
            }
            event_str = json.dumps(event, ensure_ascii=False)
            self._last_hash = event_hash
 
        self.logger.info(event_str)
 
    def log_response(self, action: str, target: str, result: str):
        """记录响应动作"""
        event = {
            'timestamp': datetime.now().isoformat(),
            'event_type': 'response',
            'action': action,
            'target': target,
            'result': result,
        }
 
        event_str = json.dumps(event, ensure_ascii=False)
 
        if self.enable_integrity:
            event_hash = self._compute_hash(event_str)
            event['integrity'] = {
                'hash': event_hash,
                'prev_hash': self._last_hash or 'genesis',
                'algorithm': 'sha256',
            }
            event_str = json.dumps(event, ensure_ascii=False)
            self._last_hash = event_hash
 
        self.logger.info(event_str)
 
    def verify_integrity(self, log_file: str = None) -> dict:
        """
        【安全修复】验证日志文件完整性
        通过重放哈希链检查是否有日志被篡改或删除
        """
        if not self.enable_integrity:
            return {'valid': True, 'message': '完整性校验未启用'}
 
        log_path = log_file or os.path.join(self.log_dir, 'security.log')
        if not os.path.exists(log_path):
            return {'valid': False, 'message': '日志文件不存在'}
 
        prev_hash = 'genesis'
        line_count = 0
        invalid_lines = []
 
        with open(log_path, 'r', encoding='utf-8') as f:
            for line_num, line in enumerate(f, 1):
                line_count += 1
                try:
                    entry = json.loads(line.strip())
                    integrity = entry.get('integrity', {})
                    expected_hash = integrity.get('hash', '')
                    entry_prev = integrity.get('prev_hash', '')
 
                    # 验证哈希链
                    if entry_prev != prev_hash:
                        invalid_lines.append(
                            f"行 {line_num}: 哈希链断裂 (期望 prev={prev_hash}, 实际 prev={entry_prev})"
                        )
 
                    # 重算哈希
                    entry_copy = {k: v for k, v in entry.items() if k != 'integrity'}
                    recomputed = self._compute_hash(json.dumps(entry_copy, ensure_ascii=False))
                    # 需要用上一条哈希重算
                    if prev_hash != 'genesis':
                        combined = f"{prev_hash}:{json.dumps(entry_copy, ensure_ascii=False)}"
                        recomputed = hashlib.sha256(combined.encode('utf-8')).hexdigest()[:16]
                    else:
                        recomputed = hashlib.sha256(
                            json.dumps(entry_copy, ensure_ascii=False).encode('utf-8')
                        ).hexdigest()[:16]
 
                    if recomputed != expected_hash:
                        invalid_lines.append(
                            f"行 {line_num}: 哈希不匹配 (期望={expected_hash}, 实际={recomputed})"
                        )
 
                    prev_hash = expected_hash
                except json.JSONDecodeError:
                    invalid_lines.append(f"行 {line_num}: JSON 解析失败")
 
        return {
            'valid': len(invalid_lines) == 0,
            'total_lines': line_count,
            'invalid_lines': invalid_lines,
        }

Step 8.1 主入口（串联所有模块）
创建 main.py：
"""
AI-Security-Guardian 主入口
串联所有模块，启动完整检测流水线
【安全修复】增加威胁情报前置过滤、容错降级、消息队列集成
"""
import time
import logging
import signal
import sys
from src.collectors.packet_collector import PacketCollector
from src.collectors.log_collector import LogCollector
from src.collectors.threat_intel import ThreatIntelCollector
from src.features.flow_features import FlowFeatureExtractor
from src.features.web_features import WebFeatureExtractor
from src.detectors.ddos_detector import DDoSDetector
from src.detectors.intrusion_detector import IntrusionDetector
from src.detectors.web_detector import WebAttackDetector
from src.detectors.anomaly_detector import AnomalyDetector
from src.decision.fusion_engine import FusionEngine
from src.response.responder import SecurityResponder
from src.audit.security_logger import SecurityLogger
 
logging.basicConfig(
    level=logging.INFO,
    format='[%(asctime)s] %(name)s - %(levelname)s - %(message)s'
)
logger = logging.getLogger('main')
 
 
class SecurityGuardian:
    """AI 安全守卫主控制器"""
 
    def __init__(self, dry_run: bool = True):
        self.dry_run = dry_run
        self._running = False
 
        # 初始化各模块
        self.security_logger = SecurityLogger(enable_integrity=True)
        self.packet_collector = PacketCollector()
        self.web_log_collector = LogCollector('/var/log/nginx/access.log', log_type='web')
 
        # 【新增】威胁情报采集器（前置过滤）
        self.threat_intel = ThreatIntelCollector(
            abuseipdb_key=os.environ.get('ABUSEIPDB_API_KEY', ''),
            virustotal_key=os.environ.get('VIRUSTOTAL_API_KEY', ''),
        )
 
        # 特征提取器
        self.flow_extractor = FlowFeatureExtractor()
        self.web_extractor = WebFeatureExtractor()
 
        # 检测引擎
        self.ddos_detector = DDoSDetector()
        self.intrusion_detector = IntrusionDetector(use_deep_learning=False)
        self.web_detector = WebAttackDetector()
        self.anomaly_detector = AnomalyDetector()
 
        # 融合决策引擎
        self.fusion_engine = FusionEngine()
 
        # 响应引擎
        self.responder = SecurityResponder(dry_run=self.dry_run)
 
    def load_models(self):
        """加载所有模型（容错：单个模型加载失败不影响整体）"""
        import os
 
        models = {
            'DDoS': ('models/saved/ddos_rf_v1.pkl', self.ddos_detector),
            '入侵检测': ('models/saved/intrusion_rf_v1.pkl', self.intrusion_detector),
            'Web攻击': ('models/saved/web_attack_nb_v1.pkl', self.web_detector),
            '异常检测': ('models/saved/anomaly_if_v1.pkl', self.anomaly_detector),
        }
 
        for name, (path, detector) in models.items():
            try:
                detector.load_model(path)
                logger.info(f"模型加载成功: {name}")
            except FileNotFoundError:
                logger.warning(f"模型未找到: {name} ({path})，请先运行训练脚本")
            except Exception as e:
                logger.error(f"模型加载失败: {name} - {e}")
 
    def process_packet(self, pkt: dict):
        """处理单个数据包"""
        src_ip = pkt.get('src_ip', '')
 
        # 【新增】威胁情报前置过滤（对应架构文档 §5.2）
        intel_result = self.threat_intel.check_ip(src_ip)
        if intel_result.get('is_malicious'):
            from src.detectors.base import DetectionResult
            result = DetectionResult(
                threat_type='threat_intel',
                threat_level='high',
                confidence=0.95,
                details=f"威胁情报命中: {intel_result.get('source', 'unknown')}",
                source_ip=src_ip,
            )
            self.responder.respond(result)
            self.security_logger.log_event(
                event_type='threat_intel', level='high',
                details={'source': intel_result.get('source')},
                source_ip=src_ip, confidence=0.95,
            )
            return
 
        # 提取流特征
        flow_features = self.flow_extractor.extract([pkt])
        for feat in flow_features:
            # 多引擎并行检测
            results = []
            if self.ddos_detector.is_ready:
                results.append(self.ddos_detector.detect(feat))
            if self.intrusion_detector.is_ready:
                results.append(self.intrusion_detector.detect(feat))
            if self.anomaly_detector.is_ready:
                results.append(self.anomaly_detector.detect(feat))
 
            # 融合决策
            final = self.fusion_engine.fuse(results)
 
            if final.threat_type != 'normal':
                self.responder.respond(final)
                self.security_logger.log_event(
                    event_type=final.threat_type,
                    level=final.threat_level,
                    details={'detection': final.details},
                    source_ip=final.source_ip,
                    confidence=final.confidence,
                )
 
    def process_web_log(self, log: dict):
        """处理 Web 日志"""
        src_ip = log.get('src_ip', '')
 
        # 威胁情报前置过滤
        intel_result = self.threat_intel.check_ip(src_ip)
        if intel_result.get('is_malicious'):
            from src.detectors.base import DetectionResult
            result = DetectionResult(
                threat_type='threat_intel',
                threat_level='high',
                confidence=0.95,
                details=f"威胁情报命中: {src_ip}",
                source_ip=src_ip,
            )
            self.responder.respond(result)
            return
 
        # Web 特征提取 + 检测
        web_feat = self.web_extractor.extract(log)
        web_result = self.web_detector.detect(web_feat)
        if web_result.threat_type != 'normal':
            self.responder.respond(web_result)
            self.security_logger.log_event(
                event_type=web_result.threat_type,
                level=web_result.threat_level,
                details={'url': log.get('url', ''), 'detection': web_result.details},
                source_ip=log.get('src_ip', ''),
                confidence=web_result.confidence,
            )
 
    def run(self):
        """主运行循环"""
        self._running = True
        logger.info("系统已启动，开始监控...")
 
        try:
            self.packet_collector.start()
        except PermissionError:
            logger.error("抓包需要管理员权限，请使用 sudo 运行")
 
        try:
            while self._running:
                # 获取数据包
                packets = self.packet_collector.get_packets(max_count=50)
                for pkt in packets:
                    self.process_packet(pkt)
 
                # 获取 Web 日志
                web_logs = self.web_log_collector.read_new_lines()
                for log in web_logs:
                    self.process_web_log(log)
 
                time.sleep(1)
 
        except KeyboardInterrupt:
            logger.info("收到停止信号，正在关闭...")
        finally:
            self.shutdown()
 
    def shutdown(self):
        """优雅关闭"""
        self._running = False
        self.packet_collector.stop()
        logger.info("系统已停止")
 
 
def main():
    import os
    logger.info("=" * 60)
    logger.info("AI-Security-Guardian 启动中...")
    logger.info("=" * 60)
 
    dry_run = os.environ.get('DRY_RUN', 'true').lower() == 'true'
    guardian = SecurityGuardian(dry_run=dry_run)
    guardian.load_models()
 
    # 信号处理（优雅关闭）
    def signal_handler(sig, frame):
        guardian.shutdown()
        sys.exit(0)
 
    signal.signal(signal.SIGINT, signal_handler)
    signal.signal(signal.SIGTERM, signal_handler)
 
    guardian.run()
 
 
if __name__ == '__main__':
    main()
